cmake_minimum_required(VERSION 3.25)
project(SwitchPrepBackend LANGUAGES C)
foreach(required HACTOOL_SOURCE OPENSSL_ROOT LZ4_SOURCE)
    if(NOT DEFINED ${required})
        message(FATAL_ERROR "Set ${required} to the corresponding source/library directory")
    endif()
endforeach()
set(parts save sha aes extkeys rsa npdm bktr kip packages pki pfs0 hfs0 nca0_romfs romfs utils nax0 nso nca xci filepath ConvertUTF cJSON storage main)
list(TRANSFORM parts PREPEND "${HACTOOL_SOURCE}/")
list(TRANSFORM parts APPEND .c)
add_executable(hactool ${parts} "${LZ4_SOURCE}/lib/lz4.c")
target_include_directories(hactool PRIVATE "${HACTOOL_SOURCE}" "${OPENSSL_ROOT}/include")
target_link_libraries(hactool PRIVATE "${OPENSSL_ROOT}/lib/libcrypto.a")
target_compile_definitions(hactool PRIVATE _BSD_SOURCE _POSIX_SOURCE _POSIX_C_SOURCE=200112L _DEFAULT_SOURCE __USE_MINGW_ANSI_STDIO=1 _FILE_OFFSET_BITS=64)
target_compile_options(hactool PRIVATE -w)
set_target_properties(hactool PROPERTIES C_STANDARD 11 C_EXTENSIONS ON)
install(TARGETS hactool RUNTIME DESTINATION bin/backend)
install(FILES "${HACTOOL_SOURCE}/LICENSE" DESTINATION share/licenses/YKW4Recomp/hactool)
if(WIN32)
    enable_language(RC)
    target_sources(hactool PRIVATE "${CMAKE_CURRENT_LIST_DIR}/hactool.rc")
    target_link_options(hactool PRIVATE -static)
    target_link_libraries(hactool PRIVATE bcrypt ws2_32 crypt32 advapi32 user32)
endif()
